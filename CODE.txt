#include <Wire.h>
#include <LiquidCrystal_I2C.h>

// LCD I2C address 0x27 with 16 columns and 2 rows
LiquidCrystal_I2C lcd(0x27, 16, 2);

// Ultrasonic sensor pins
const int trigPin = 7;
const int echoPin = 6;

// RGB LED pins
const int redPin   = 11;
const int bluePin  = 10;
const int greenPin = 9;

// Buzzer pin
const int buzzerPin = 8;

void setup() {

  // Ultrasonic sensor setup
  pinMode(trigPin, OUTPUT);
  pinMode(echoPin, INPUT);

  // RGB LED setup
  pinMode(redPin, OUTPUT);
  pinMode(greenPin, OUTPUT);
  pinMode(bluePin, OUTPUT);

  // Buzzer setup
  pinMode(buzzerPin, OUTPUT);

  // LCD initialization
  lcd.init();
  lcd.backlight();

  // Welcome message
  lcd.setCursor(0, 0);
  lcd.print("Parking Assist");

  delay(2000);

  lcd.clear();
}

void loop() {

  // Send ultrasonic pulse
  digitalWrite(trigPin, LOW);
  delayMicroseconds(2);

  digitalWrite(trigPin, HIGH);
  delayMicroseconds(10);

  digitalWrite(trigPin, LOW);

  // Read echo duration
  long duration = pulseIn(echoPin, HIGH, 30000);

  // Convert duration to distance in cm
  int distance = duration * 0.034 / 2;

  lcd.setCursor(0, 0);

  // No object detected
  if (distance == 0 || distance > 300) {

    lcd.print("Dist: --- cm    ");

    setColor(0, 0, 0);

    noTone(buzzerPin);
  }

  else {

    // Display distance
    lcd.print("Dist: ");
    lcd.print(distance);
    lcd.print(" cm    ");

    lcd.setCursor(0, 1);

    // Safe distance
    if (distance > 100) {

      lcd.print("Safe         ");

      // Green color
      setColor(0, 255, 0);

      noTone(buzzerPin);
    }

    // Warning distance
    else if (distance > 20) {

      lcd.print("Warning!     ");

      // Yellow color
      setColor(255, 255, 0);

      // Slow beep
      tone(buzzerPin, 1000);

      delay(distance * 2);

      noTone(buzzerPin);

      delay(distance * 2);
    }

    // Very close object
    else {

      lcd.print("STOP!        ");

      // Red color
      setColor(255, 0, 0);

      // Continuous buzzer
      tone(buzzerPin, 1000);
    }
  }

  delay(100);
}

// Function to control RGB LED color
void setColor(int red, int green, int blue) {

  analogWrite(redPin, red);

  analogWrite(greenPin, green);

  analogWrite(bluePin, blue);
}


/*
README

Components Used:
- Arduino Uno
- Ultrasonic Sensor (HC-SR04)
- LCD 16x2 with I2C
- RGB LED
- Piezo Buzzer
- Resistors


Connections:

Ultrasonic Sensor:
VCC  -> 5V
GND  -> GND
Trig -> Pin 7
Echo -> Pin 6


RGB LED:
Red pin    -> Pin 11
Green pin  -> Pin 9
Blue pin   -> Pin 10
Common pin -> GND


Buzzer:
Positive -> Pin 8
Negative -> GND


LCD I2C:
VCC -> 5V
GND -> GND
SDA -> A4
SCL -> A5


How to Run:
1. Open the project in Tinkercad.
2. Start Simulation.
3. Move an object closer or farther from the ultrasonic sensor.
4. Watch the LCD, RGB LED, and buzzer behavior.


Expected Result:

If no object is detected:
- LCD shows: Dist: --- cm
- RGB LED OFF
- Buzzer OFF


If object distance is more than 100 cm:
- LCD shows: Safe
- Green light ON
- Buzzer OFF


If object distance is between 20 cm and 100 cm:
- LCD shows: Warning!
- Yellow light ON
- Buzzer beeps slowly


If object distance is less than or equal to 20 cm:
- LCD shows: STOP!
- Red light ON
- Continuous buzzer sound

*/
