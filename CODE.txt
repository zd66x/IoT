// --- Pin Definitions ---
const int sensorPin = A0;   // TMP36 connected to analog pin A0
const int greenLED  = 8;    // Green LED on pin 8
const int redLED    = 9;    // Red LED on pin 9
const int buzzerPin = 10;   // Buzzer on pin 10

// --- Alert Threshold ---
const float threshold = 30.0;  // Alert threshold in Celsius

void setup() {
  // Initialize digital pins as outputs
  pinMode(greenLED, OUTPUT);
  pinMode(redLED, OUTPUT);
  pinMode(buzzerPin, OUTPUT);
  
  // Initialize serial communication for monitoring
  Serial.begin(9600);
  Serial.println("Smart Temperature Alert System - Started");
}

void loop() {
  // Read analog value from TMP36 sensor (0 - 1023)
  int rawValue = analogRead(sensorPin);
  
  // Convert raw ADC value to voltage (5V reference)
  float voltage = rawValue * (5.0 / 1023.0);
  
  // Convert voltage to Celsius
  // TMP36 offset is 500mV, scale factor is 10mV/degree
  float tempC = (voltage - 0.5) * 100.0;

  // Print temperature to Serial Monitor
  Serial.print("Temperature: ");
  Serial.print(tempC);
  Serial.println(" C");

  // Check temperature against threshold
  if (tempC < threshold) {
    // Normal state: Temperature is below threshold
    digitalWrite(greenLED, HIGH);  // Turn ON green LED
    digitalWrite(redLED, LOW);     // Turn OFF red LED
    noTone(buzzerPin);             // Turn OFF buzzer
  } else {
    // Alert state: Temperature is above or equal to threshold
    digitalWrite(greenLED, LOW);   // Turn OFF green LED
    digitalWrite(redLED, HIGH);    // Turn ON red LED
    tone(buzzerPin, 1000);         // Sound buzzer at 1kHz
  }

  // Wait 1 second before next reading
  delay(1000);
}
