const int sensorPin = A0;   // TMP36 connected to analog pin A0
const int greenLED  = 8;    // Green LED on pin 8
const int redLED    = 9;    // Red LED on pin 9
const int buzzerPin = 10;   // Buzzer on pin 10

const float threshold = 30.0;  // Alert threshold in Celsius

void setup() {

  pinMode(greenLED, OUTPUT);
  pinMode(redLED, OUTPUT);
  pinMode(buzzerPin, OUTPUT);

  Serial.begin(9600);

  Serial.println("Smart Temperature Alert System - Started");
}

void loop() {

  int rawValue = analogRead(sensorPin);
  
  // Convert raw value to voltage
  float voltage = rawValue * (5.0 / 1023.0);

  // Convert voltage to temperature
  float tempC = (voltage - 0.5) * 100.0;

  Serial.print("Temperature: ");
  Serial.print(tempC);
  Serial.println(" C");

  // Normal condition
  if (tempC < threshold) {

    digitalWrite(greenLED, HIGH);
    digitalWrite(redLED, LOW);

    noTone(buzzerPin);

  }

  // High temperature condition
  else {

    digitalWrite(greenLED, LOW);
    digitalWrite(redLED, HIGH);

    tone(buzzerPin, 1000);
  }

  delay(1000);
}



README
-------------------------
Components Used:
- Arduino Uno
- TMP36 Temperature Sensor
- Green LED
- Red LED
- Piezo Buzzer
- 2 Resistors (220 ohm)

Connections:
TMP36:
Left pin  -> 5V
Middle pin -> A0
Right pin -> GND

Green LED:
Positive -> Pin 8 through resistor
Negative -> GND

Red LED:
Positive -> Pin 9 through resistor
Negative -> GND

Buzzer:
Positive -> Pin 10
Negative -> GND


How to Run:
1. Open the project in Tinkercad.
2. Start Simulation.
3. Click on TMP36 sensor.
4. Change the temperature value.

Expected Result:
- If temperature is below 30C:
  Green LED turns ON.

- If temperature is 30C or higher:
  Red LED and buzzer turn ON.
